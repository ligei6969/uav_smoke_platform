cmake_minimum_required(VERSION 3.10)
project(m350_smoke_task VERSION 0.1.0 LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build")
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SMOKE_WITH_PSDK_ADAPTER "Compile field conversions against PSDK headers (not a live hardware driver)" OFF)
option(SMOKE_WITH_VIDEO "Build the Linux OpenCV CPU recorded-video pipeline" OFF)
option(SMOKE_ASAN "Enable address+undefined sanitizers on Linux GCC/Clang" OFF)
option(SMOKE_TSAN "Enable thread sanitizer on Linux GCC/Clang" OFF)
if(SMOKE_ASAN AND SMOKE_TSAN)
    message(FATAL_ERROR "ASan and TSan require separate builds")
endif()
find_package(Threads REQUIRED)
add_library(smoke_core
    src/core/types.cpp
    src/replay/replay_io.cpp
    src/telemetry/telemetry.cpp
    src/localization/localization.cpp
    src/gimbal/gimbal.cpp
    src/events/stable_event.cpp
    src/events/evidence_store.cpp
    src/video/h264_stream.cpp
)
target_include_directories(smoke_core PUBLIC include)
target_link_libraries(smoke_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(smoke_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(smoke_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SMOKE_ASAN OR SMOKE_TSAN)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer builds require Linux GCC/Clang")
    endif()
    if(SMOKE_TSAN)
        set(SMOKE_SANITIZERS "thread")
    else()
        set(SMOKE_SANITIZERS "address,undefined")
    endif()
    target_compile_options(smoke_core PUBLIC -fsanitize=${SMOKE_SANITIZERS} -fno-omit-frame-pointer -g)
    target_link_libraries(smoke_core PUBLIC -fsanitize=${SMOKE_SANITIZERS})
endif()
add_executable(smoke_replay src/app/main.cpp)
target_link_libraries(smoke_replay PRIVATE smoke_core)

if(SMOKE_WITH_VIDEO)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The initial video target is validated on Linux only; leave SMOKE_WITH_VIDEO=OFF for the portable core")
    endif()
    find_package(OpenCV 4.6 REQUIRED COMPONENTS core imgproc imgcodecs videoio dnn)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SMOKE_FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
    add_library(smoke_video
        src/video/model.cpp
        src/video/report.cpp
        src/video/pipeline.cpp)
    target_include_directories(smoke_video PUBLIC ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(smoke_video PUBLIC smoke_core ${OpenCV_LIBS} OpenSSL::Crypto PkgConfig::SMOKE_FFMPEG)
    target_compile_options(smoke_video PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(smoke_analyze_video src/app/analyze_video.cpp)
    target_link_libraries(smoke_analyze_video PRIVATE smoke_video)
    add_executable(smoke_model_probe src/video/model_probe.cpp)
    target_link_libraries(smoke_model_probe PRIVATE smoke_video)
    install(TARGETS smoke_analyze_video RUNTIME DESTINATION bin)
endif()

if(SMOKE_WITH_PSDK_ADAPTER)
    set(PSDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Unmodified DJI PSDK root")
    if(NOT EXISTS "${PSDK_ROOT}/psdk_lib/include/dji_fc_subscription.h")
        message(FATAL_ERROR "PSDK_ROOT must contain psdk_lib/include/dji_fc_subscription.h")
    endif()
    add_library(smoke_psdk_mapping src/adapters/dji_telemetry.cpp)
    target_include_directories(smoke_psdk_mapping PUBLIC "${PSDK_ROOT}/psdk_lib/include")
    target_link_libraries(smoke_psdk_mapping PUBLIC smoke_core)
    message(STATUS "PSDK field adapter enabled: headers only; no aircraft connection or SDK activation")
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(name queue localization gimbal io events h264_stream)
        add_executable(test_${name} tests/test_${name}.cpp)
        target_link_libraries(test_${name} PRIVATE smoke_core)
        add_test(NAME ${name} COMMAND test_${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endforeach()
    if(SMOKE_WITH_PSDK_ADAPTER)
        add_executable(test_psdk_mapping tests/test_psdk_mapping.cpp)
        target_link_libraries(test_psdk_mapping PRIVATE smoke_psdk_mapping)
        add_test(NAME psdk_mapping COMMAND test_psdk_mapping)
        set_tests_properties(psdk_mapping PROPERTIES TIMEOUT 30)
    endif()
    # Ubuntu 18.04 on the original Nano ships CMake 3.10, before FindPython3.
    if(CMAKE_VERSION VERSION_LESS "3.12")
        find_package(PythonInterp 3.6 REQUIRED)
        set(SMOKE_TEST_PYTHON "${PYTHON_EXECUTABLE}")
    else()
        find_package(Python3 3.6 COMPONENTS Interpreter REQUIRED)
        set(SMOKE_TEST_PYTHON "${Python3_EXECUTABLE}")
    endif()
    add_test(NAME replay_integration COMMAND "${SMOKE_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_replay.py" $<TARGET_FILE:smoke_replay>)
    set_tests_properties(replay_integration PROPERTIES TIMEOUT 30)
    add_test(NAME detection_export COMMAND "${SMOKE_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_export_detections.py")
    set_tests_properties(detection_export PROPERTIES TIMEOUT 30)
    if(SMOKE_WITH_VIDEO)
        foreach(name video_model video_report)
            add_executable(test_${name} tests/test_${name}.cpp)
            target_link_libraries(test_${name} PRIVATE smoke_video)
            add_test(NAME ${name} COMMAND test_${name})
            set_tests_properties(${name} PROPERTIES TIMEOUT 30)
        endforeach()
        set(SMOKE_TEST_MODEL "${CMAKE_CURRENT_SOURCE_DIR}/../samples/sample_c++/module_sample/liveview/data/best.onnx" CACHE FILEPATH "Real model for opt-in CLI integration tests")
        if(EXISTS "${SMOKE_TEST_MODEL}")
            add_test(NAME video_cli COMMAND "${SMOKE_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_video_cli.py" $<TARGET_FILE:smoke_analyze_video> "${SMOKE_TEST_MODEL}")
            set_tests_properties(video_cli PROPERTIES TIMEOUT 120)
        endif()
    endif()
endif()
install(TARGETS smoke_replay RUNTIME DESTINATION bin)
