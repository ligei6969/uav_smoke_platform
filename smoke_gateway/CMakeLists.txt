cmake_minimum_required(VERSION 3.16)
project(smoke_gateway VERSION 0.1.0 LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build")
endif()
option(SMOKE_ASAN "Enable Address/UndefinedBehavior sanitizers (Linux GCC/Clang)" OFF)
option(SMOKE_TSAN "Enable ThreadSanitizer (Linux GCC/Clang)" OFF)
if(SMOKE_ASAN AND SMOKE_TSAN)
    message(FATAL_ERROR "ASan and TSan require separate builds")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(smoke_core
    src/core.cpp
    src/journal.cpp
    src/protocol.cpp
)
target_include_directories(smoke_core PUBLIC include)
target_link_libraries(smoke_core PUBLIC Threads::Threads)
add_executable(smoke_gateway src/server.cpp)
target_link_libraries(smoke_gateway PRIVATE smoke_core)
if(WIN32)
    target_link_libraries(smoke_gateway PRIVATE ws2_32)
endif()
foreach(target smoke_core smoke_gateway)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(SMOKE_ASAN OR SMOKE_TSAN)
    if(NOT UNIX OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer builds require Linux GCC/Clang")
    endif()
    if(SMOKE_TSAN)
        set(sanitizer thread)
    else()
        set(sanitizer address,undefined)
    endif()
    target_compile_options(smoke_core PUBLIC -fsanitize=${sanitizer} -fno-omit-frame-pointer -g)
    target_link_options(smoke_core PUBLIC -fsanitize=${sanitizer})
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(core_test tests/core_test.cpp)
    target_link_libraries(core_test PRIVATE smoke_core)
    # Standalone checks must remain active in Release builds.
    if(MSVC)
        target_compile_options(core_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(core_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME core COMMAND core_test)
    set_tests_properties(core PROPERTIES TIMEOUT 60)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py --server $<TARGET_FILE:smoke_gateway>)
    set_tests_properties(network PROPERTIES TIMEOUT 90)
    add_test(NAME client COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_client.py -v)
    set_tests_properties(client PROPERTIES TIMEOUT 30)
endif()
install(TARGETS smoke_gateway RUNTIME DESTINATION bin)
